cmake_minimum_required(VERSION 3.8)
project(humanoid_camera)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME} PACKAGE_DIR python/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/timestamp_adapter.py scripts/check_cameras.py scripts/gain_controller.py scripts/collect_camera_diagnostics.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS start_camera.sh start_cameras.sh DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY docs DESTINATION share/${PROJECT_NAME})
ament_package()
